# ===================================================================
# 虚拟触屏按键 / 虚拟鼠标（Minecraft on ART 移植）
#   路径：/sdcard/.minecraft/vcontrols.txt
#   生效：改完重启游戏（布局是 native 运行时算的，不需要重编 APK）
#   语法：key=value，每行一条；# 后为注释；解析失败自动退回默认值
#
# 【坐标规范与 FCL（Fold Craft Launcher）一致】
#   · 所有坐标是「千分比」(per-mille)：0~1000，不是 0~1；
#   · 按钮位置 x,y：左/上边缘在“可用空间”里的千分比
#       像素 x = (屏宽 - 按钮宽) * x / 1000     （x=0 贴左，x=1000 贴右）
#       像素 y = (屏高 - 按钮高) * y / 1000
#   · 按钮宽高 w,h：千分比，但基准是「屏幕高度」（FCL 的 reference:SCREEN_HEIGHT）
#       像素 宽 = 屏高 * w / 1000
#   · 视角区 = 整屏减去所有按钮和摇杆矩形（FCL 默认没有“视角按钮”）
# ===================================================================

# 总开关：0 = 完全关闭（恢复改动前的行为，可见浮层也一起消失），1 = 启用
# 提示：现在屏幕上有半透明浮层标出每个按键的位置（Java View 画的，不吃触摸），
#       进菜单时 JUMP/DROP/CHAT/摇杆会自动隐藏（FCL 的 IN_GAME 语义）。
enabled=1

# 视角灵敏度倍数。手机滑动通常 1.5~2.5 手感更好
look_sensitivity=1.0

# 上下反转：0 = 手指下滑视角下移（推荐），1 = 反转
invert_y=0

# 调试日志：1 = 每次触摸/按键都往 logcat 打一条（默认 0，只打首次）
debug=0

# 游戏内「视角区轻点」= 左键单击（挖掘/攻击一次）。
# 1 = 开（默认，推荐）：没有可见浮层时，点哪都能挖一下，和手机版 MC 的习惯一致。
# 0 = 关：轻点什么都不做，只能靠右下角的 × 按钮攻击。
# 注意：拖动永远是转视角，不会误触攻击；要连续挖就按住 × 区域。
tap_attack=1

# -------------------------------------------------------------------
# 移动控件：dpad = 十字键（FCL 的 BUTTON 风格，默认），stick = 摇杆（FCL 的 ROCKER 风格）
#   FCL 源码里 ControlDirectionStyle.java:58 的默认值就是 BUTTON，
#   默认布局 directionStyles[0].styleType 也是 "BUTTON"，所以这里默认 dpad。
#   摇杆实现整套保留，改 move_mode=stick 即切回。
# -------------------------------------------------------------------
move_mode=dpad

# 十字键布区：dpad_zone=x,y,w,h（千分比，宽高以屏幕高为基准）
#   默认沿用 FCL 摇杆的足迹 50,900,450,450（左下角）
dpad_zone=50,900,450,450

# 四个方向键之间的间隙（千分比，FCL 的 buttonStyle.interval 默认 50）
dpad_interval=50

# 允许斜向：1 = 相邻两个方向可同时按住（两只手指 W+D 斜跑，MC 移动的刚需）
dpad_diag=1

# 中心格 = 潜行（FCL 的 sneak:true / sneakKeycode:42），1 开 0 关
dpad_sneak_center=1

# 潜行用的 GLFW 键码（340 = GLFW_KEY_LEFT_SHIFT）
dpad_sneak_key=340

# -------------------------------------------------------------------
# 左下摇杆（仅 move_mode=stick 时生效）：stick=x,y,size
#   x,y  千分比位置        size 边长/屏幕高度千分比
#   FCL 默认 50,900,450；按 FCL 语义“按下点即圆心”(CENTER_FOLLOW)
# -------------------------------------------------------------------
stick=50,900,450

# 双击方向键锁定潜行（FCL 的 sneak:true）；0 = 关闭
stick_sneak=1

# -------------------------------------------------------------------
# 可选：把视角拖动限制在某块区域（千分比 x0,y0,x1,y1）
# 默认不写 = 除按钮/摇杆以外的整屏都能转视角
# -------------------------------------------------------------------
# look_zone=450,0,1000,1000

# -------------------------------------------------------------------
# 可选：单独覆盖某个按钮（一般不用改）
#   btn_<名字>=x,y,w,h    千分比；把 w 或 h 写 0 就是禁用该键
#
# 默认布局（直接搬 FCL 的 00000000.json）：
#   名字       文字  x    y     w    h     作用
#   ATTACK     ×    800  750   140  140    左键（挖掘/攻击）
#   JUMP       ■    875  750   140  140    空格（跳）
#   USE        ○    875  565   140  140    右键（放置/使用）
#   INV        E    720 1000   100  100    背包
#   SNEAK      ▼    950  935   140  140    潜行（Shift）
#   DROP       Q   1000  400   120  120    丢弃
#   ESC        Esc    0    0   160  120    菜单
#   F3         F3   100    0   160  120    调试屏
#   CHAT       T    200    0   160  120    聊天
#   ENTER      ↵    900    0   160  120    回车
#   BACKSP     ⌫   1000    0   160  120    退格
#
# 默认禁用（w=h=0），需要时给个位置就打开：
#   WHEELUP ▲ / WHEELDN ▼（物品栏）、KEY_W/KEY_A/KEY_S/KEY_D（备用虚拟键）
# -------------------------------------------------------------------
# 例：把跳跃键放大并挪到右下角
# btn_JUMP=880,790,180,180
# 注：DROP 默认已从 x=1000（贴右缘）改成 940 —— 实测 1000 会被系统
#     「右边缘返回手势」吃掉（点不中），940 可正常命中。要再往里挪就写：
# btn_DROP=900,400,120,120
# 例：打开物品栏滚轮键（攻击键上方）
# btn_WHEELUP=800,620,120,120
# 例：禁用丢弃键
# btn_DROP=0,0,0,0

# ── 热键栏（快捷栏）──────────────────────────────────────────────
# 手机上没有滚轮/数字键，所以做了一排槽位：点第 N 槽 = 注入 GLFW_KEY_N（切到第 N 格）。
# 手指在槽位条上滑动会连续换槽（像 PC 滚轮）。
hotbar=1
# hotbar_zone=x,y,w,h（千分比；宽高以屏幕高为基准，语义同 dpad_zone）
#   默认放在底部中央并**留出底边距**：贴边会被系统返回/上滑手势吃掉。
hotbar_zone=300,855,400,52
# 槽位数（1~9）；9 与 MC 的快捷栏一致
hotbar_slots=9
